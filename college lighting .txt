public with sharing class CollegeController {

    @AuraEnabled
    public static String saveStudent(

        String studentName,
        Integer rollNo,
        Decimal marks,
        String email

    ) {

        if(studentName == null || studentName == '') {

            return 'Student Name cannot be blank';
        }

        if(rollNo <= 0) {

            return 'Roll Number must be greater than 0';
        }

        if(marks < 0 || marks > 100) {

            return 'Marks should be between 0 and 100';
        }

        if(!email.contains('@')) {

            return 'Invalid Email';
        }

        Student__c s = new Student__c();

        s.Name = studentName;
        s.Roll_Number__c = rollNo;
        s.Marks__c = marks;
        s.Email__c = email;

        insert s;

        return 'Student Record Saved Successfully';
    }



    @AuraEnabled
    public static String saveFaculty(

        String facultyName,
        Integer facultyId,
        Decimal salary,
        String department,
        Date joiningDate

    ) {

        if(facultyName == null || facultyName.length() < 3) {

            return 'Faculty Name must contain at least 3 characters';
        }

        if(facultyId <= 0) {

            return 'Faculty ID must be greater than 0';
        }

        if(salary < 10000 || salary > 500000) {

            return 'Salary must be between 10000 and 500000';
        }

        if(joiningDate > Date.today()) {

            return 'Joining Date cannot be future date';
        }

        Faculty__c f = new Faculty__c();

        f.Name = facultyName;
        f.Faculty_ID__c = facultyId;
        f.Salary__c = salary;
        f.Department__c = department;
        f.Joining_Date__c = joiningDate;

        insert f;

        return 'Faculty Record Saved Successfully';
    }
}

*/
