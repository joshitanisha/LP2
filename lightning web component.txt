/*

public with sharing class EmployeeController {

    @AuraEnabled
    public static String saveEmployee(
        String name,
        Integer empId,
        String email,
        Decimal salary,
        String department,
        Date joiningDate
    ) {

        if(name == null || name.length() < 3) {
            return 'Employee Name must contain at least 3 characters';
        }

        if(empId <= 0) {
            return 'Employee ID must be greater than 0';
        }

        if(salary < 10000 || salary > 500000) {
            return 'Salary must be between 10000 and 500000';
        }

        if(joiningDate > Date.today()) {
            return 'Joining Date cannot be future date';
        }

        Employee__c emp = new Employee__c();

        emp.Name = name;
        emp.Employee_ID__c = empId;
        emp.Email__c = email;
        emp.Salary__c = salary;
        emp.Department__c = department;
        emp.Joining_Date__c = joiningDate;

        insert emp;

        return 'Employee Saved Successfully';
    }
}

/*

import { LightningElement } from 'lwc';
import saveEmployeeData from '@salesforce/apex/EmployeeController.saveEmployee';

export default class EmployeeForm extends LightningElement {

    name;
    empId;
    email;
    salary;
    department;
    joiningDate;

    departmentOptions = [
        { label: 'HR', value: 'HR' },
        { label: 'IT', value: 'IT' },
        { label: 'Sales', value: 'Sales' },
        { label: 'Marketing', value: 'Marketing' }
    ];

    handleName(event) {
        this.name = event.target.value;
    }

    handleId(event) {
        this.empId = event.target.value;
    }

    handleEmail(event) {
        this.email = event.target.value;
    }

    handleSalary(event) {
        this.salary = event.target.value;
    }

    handleDepartment(event) {
        this.department = event.target.value;
    }

    handleDate(event) {
        this.joiningDate = event.target.value;
    }

    saveEmployee() {

        saveEmployeeData({
            name: this.name,
            empId: parseInt(this.empId),
            email: this.email,
            salary: parseFloat(this.salary),
            department: this.department,
            joiningDate: this.joiningDate
        })
        .then(result => {
            alert(result);
        })
        .catch(error => {
            alert(error.body.message);
        });
    }
}

*/
